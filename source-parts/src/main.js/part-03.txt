function beginBattle(level = 1) {
  if (!api.token()) return;

  game.remote = false;
  Promise.resolve(game.startRun(level)).catch((error) => ui.toast(error.message || "Бой не запустился"));
}

function coopSocket(msg) {
  const ws = new WebSocket(`ws://${location.hostname}:8090`);
  ws.onopen = () => ws.send(JSON.stringify(msg));
  ws.onmessage = (event) => {
    const data = JSON.parse(event.data);
    if (data.t === "err") {
      ui.toast(data.error || "Комната не найдена");
      return;
    }
    if (data.t === "room") {
      game.coop = { ws, seat: data.seat };
      $("coop-code").textContent = data.code;
      $("coop-wait").textContent = data.seat === 0 ? "Ждём друга. Пусть введёт этот код." : "Входим в уровень…";
    }
    if (data.t === "go") {
      game.coop = { ...(game.coop || {}), ws, seat: data.seat };
      game.beginCoop(data);
    } else if (data.t === "in" || data.t === "pick" || data.t === "reroll" || data.t === "continue" || data.t === "endless" || data.t === "exit") {
      game.onCoop(data);
    }
  };
  ws.onerror = () => ui.toast("Нет связи с комнатой");
}

const PART_COLLECTION = ["gun", "drone", "laser", "scatter", "grenade", "emp", "orb", "common"];
let collectionFrom = "tasks";

function collectionKey(item) {
  return `${item.kind}:${item.id}`;
}

function collectionCatalog() {
  const seen = meta.seen || {};
  const enemies = new Set(seen.enemies || []);
  const branches = new Set(seen.branches || []);
  const synergies = new Set(seen.synergies || []);
  const partFamilies = new Set((profile?.parts || []).map((p) => p.family || "common"));
  return [
    ...Object.entries(SHAPES).map(([id, shape]) => ({
      kind: "enemy", id, section: "Враги", label: shape.name, glyph: shape.glyph, unlocked: enemies.has(id),
    })),
    ...Object.entries(BRANCH_LABELS).map(([id, label]) => ({
      kind: "branch", id, section: "Ветки оружия", label, glyph: "", unlocked: branches.has(id),
    })),
    ...SYNERGIES.map((synergy) => ({
      kind: "synergy", id: synergy.id, section: "Синергии", label: synergy.name, glyph: "", unlocked: synergies.has(synergy.id),
    })),
    ...PART_COLLECTION.map((id) => ({
      kind: "part",
      id,
      section: "Семейства запчастей",
      label: partFamilyLabel(id === "common" ? null : id),
      glyph: "",
      unlocked: partFamilies.has(id),
    })),
  ];
}

function collectionSnapshot() {
  const items = collectionCatalog();
  const unlocked = items.filter((item) => item.unlocked).map(collectionKey);
  if (!Array.isArray(meta.collectionViewed)) {
    meta.collectionViewed = collectionNews(unlocked, undefined).viewed;
    saveMeta(meta);
  }
  const news = collectionNews(unlocked, meta.collectionViewed);
  return { items, fresh: new Set(news.fresh) };
}

function paintCollectionHub(snapshot = collectionSnapshot()) {
  const open = snapshot.items.filter((item) => item.unlocked).length;
  const labels = snapshot.items
    .filter((item) => snapshot.fresh.has(collectionKey(item)))
    .map((item) => item.label);
  const line = formatFresh(labels);
  $("collection-progress").textContent = `Открыто ${open} из ${snapshot.items.length}`;
  const freshEl = $("collection-fresh");
  freshEl.textContent = line ? `Новое: ${line}` : "";
  freshEl.classList.toggle("hidden", !line);
  const notice = $("btn-collection-notice");
  notice.textContent = line ? `Новое в коллекции: ${line}` : "";
  notice.classList.toggle("hidden", !line);
}

function collectionChip(item, fresh) {
  const glyphHtml = item.glyph ? `<span class="coll-glyph">${item.glyph}</span>` : "";
  const cls = `${item.unlocked ? "" : " locked"}${fresh ? " fresh" : ""}`;
  return `<div class="coll-chip${cls}">${glyphHtml}<span>${item.unlocked ? item.label : "???"}</span></div>`;
}

function collectionSection(title, items, fresh) {
  const unlocked = items.filter((item) => item.unlocked).length;
  const chips = items.map((item) => collectionChip(item, fresh.has(collectionKey(item)))).join("");
  return `<div class="coll-section"><h3>${title} <span class="sub">${unlocked}/${items.length}</span></h3><div class="coll-grid">${chips}</div></div>`;
}

function renderCollection(items, fresh) {
  const sections = [];
  for (const item of items) {
    let section = sections.find((row) => row.title === item.section);
    if (!section) {
      section = { title: item.section, items: [] };
      sections.push(section);
    }
    section.items.push(item);
  }
  $("collection-sub").textContent = fresh.size
    ? "Новое подсвечено."
    : "Что ты уже открыл в бою.";
  $("collection-body").innerHTML = sections.map((section) => collectionSection(section.title, section.items, fresh)).join("");
}

function openCollection(from) {
  collectionFrom = from || "tasks";
  const snapshot = collectionSnapshot();
  openScreen("screen-collection");
  renderCollection(snapshot.items, snapshot.fresh);
  meta.collectionViewed = snapshot.items.filter((item) => item.unlocked).map(collectionKey);
  ui.save();
  updateBadges();
}

async function boot() {
  if (!api.token()) {
    try {
      const guest = await api.guest();
      applyProfile(guest.profile);
      showMenu();
      return;
    } catch (error) {
      showLogin(error.message || "");
      return;
    }
  }
  try {
    const me = await api.me();
    applyProfile(me.profile);
    showMenu();
  } catch (error) {
    showLogin(error.message || "Сессия не найдена");
  }
}
