function weaponIdOf(who) {
  return Object.entries(WEAPON_INFO).find(([, info]) => info.name === who)?.[0] || null;
}

function weaponLevel(run, id) {
  if (!id || !run?.weapons?.[id]) return 0;
  return 1 + (run.pips?.[id] || []).length;
}

function starRow(n) {
  if (!n) return "";
  let html = `<span class="stars">`;
  for (let i = 0; i < n; i++) html += `<i class="on"></i>`;
  return `${html}</span>`;
}

function cardIcon(card) {
  return CARD_ICONS[weaponIdOf(card.who)] || CARD_ICONS.general;
}

function loadoutSlots(run) {
  const owned = Object.keys(run?.weapons || {}).filter((id) => run.weapons[id] && WEAPON_INFO[id]);
  let html = "";
  for (let i = 0; i < 5; i++) {
    const id = owned[i];
    if (!id) {
      html += `<div class="slot empty"></div>`;
      continue;
    }
    const info = WEAPON_INFO[id];
    html += `<div class="slot" style="color:${info.color}">${CARD_ICONS[id]}<b>${info.name}</b>${starRow(weaponLevel(run, id))}</div>`;
  }
  return html;
}

function currentHordeRule() {
  return hordeRuleById(profile?.hordeRule) || hordeRuleFor(moscowDay());
}

function paintHordeRule() {
  const rule = currentHordeRule();
  const menu = $("menu-horde-rule");
  if (menu) menu.textContent = `Орда сегодня: ${rule.title}. ${rule.text}`;
  const clear = $("clear-horde-rule");
  if (clear) clear.textContent = `${rule.title}. ${rule.text}`;
}

  const track = profile?.accountTrack?.length ? profile.accountTrack : accountTrack(level);
  $("account-level-label").textContent = `Ур. ${level}`;
  $("account-xp").textContent = `${xp} / ${need}`;
  const pct = need > 0 ? Math.max(0, Math.min(100, (xp / need) * 100)) : 0;
  $("account-xp-fill").style.width = `${pct}%`;
  const nextBits = accountRewardBits(track[0]);
  $("account-next").textContent = nextBits.length ? `Дальше: ${nextBits.join(", ")}` : "";
  $("account-steps").innerHTML = track
    .map((step, index) => {
      con
