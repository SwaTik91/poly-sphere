const RARITY_ORDER = ["common", "rare", "epic", "legendary"];

function pack(specs) {
  const tiers = {};
  RARITY_ORDER.forEach((rarity, index) => {
    tiers[rarity] = specs[index];
  });
  return tiers;
}

function line(id, name, text) {
  return { id, name, text };
}

function dmgPart(id, name, key, label) {
  const pcts = [8, 15, 25, 40];
  return {
    id,
    name,
    tiers: pack(
      pcts.map((pct) => ({
        lines: [line("dmg", label, `+${pct}%`)],
        bonus: { [key]: pct / 100 },
      })),
    ),
  };
}

function branchPart(id, name, when, specs) {
  return { id, name, when, tiers: pack(specs) };
}

function num(n) {
  return formatAffixNumber(n).replace(".", ",");
}

export function formatAffixNumber(value) {
  const rounded = Math.round(value * 100) / 100;
  let text = rounded.toFixed(2);
  text = text.replace(/(\.\d*?)0+$/, "$1").replace(/\.$/, "");
  return text;
}

function findDef(family, base) {
  const list = PARTS[family || "common"] || [];
  return list.find((part) => part.id === base) || null;
}

export function partAffixes(family, base, rarity) {
  const tier = findDef(family, base)?.tiers?.[rarity];
  if (!tier) return null;
  const step = RARITY_STEP[rarity] || 1;
  return tier.lines.map((item) => ({ id: item.id, name: item.name, text: item.text, step }));
}

export function polishLevel(value) {
  const n = Math.floor(Number(value) || 0);
  if (n <= 0) return 0;
  return Math.min(POLISH_MAX, n);
}

export function scalePolishedNumber(value, level) {
  const lv = polishLevel(level);
  if (!lv || !Number.isFinite(value) || value === 0) return value;
  const scaled = value * (1 + 0.08 * lv);
  if (Number.isInteger(value)) {
    const rounded = Math.round(scaled);
    if (rounded === value) return value + Math.sign(value);
    return rounded;
  }
  const rounded = Math.round(scaled * 100) / 100;
  if (rounded === value) {
    const step = value > 0 ? 0.01 : -0.01;
    return Math.round((value + step) * 100) / 100;
  }
  return rounded;
}

const STRENGTH_KEYS = new Set(["haste", "guardHaste", "lungeHaste", "empSlowMul"]);

export function scaleBonusValue(key, value, level) {
  const lv = polishLevel(level);
  if (!lv || typeof value !== "number") return value;
  if (STRENGTH_KEYS.has(key) && value > 0 && value < 1) {
    const pct = Math.round((1 - value) * 100);
    const next = scalePolishedNumber(pct, lv);
    return Math.round((1 - next / 100) * 1000) / 1000;
  }
  if (key === "centerMul" && value >= 1) {
    const pct = Math.round((value - 1) * 100);
    const next = scalePolishedNumber(pct, lv);
    return 1 + next / 100;
  }
  return scalePolishedNumber(value, lv);
}

function scaleBonusTree(bonus, level) {
  if (!bonus || typeof bonus !== "object") return bonus;
  const out = Array.isArray(bonus) ? [] : {};
  for (const [key, value] of Object.entries(bonus)) {
    if (value && typeof value === "object") out[key] = scaleBonusTree(value, level);
    else out[key] = scaleBonusValue(key, value, level);
  }
  return out;
}

export function polishedText(text, level) {
  if (!polishLevel(level) || !text) return text || "";
  return String(text).replace(/\d+(?:[.,]\d+)?/g, (raw) => {
    const n = Number(raw.replace(",", "."));
    if (!Number.isFinite(n)) return raw;
    return formatAffixNumber(scalePolishedNumber(n, level)).replace(".", ",");
  });
}

function rollRarity(rng) {
  const r = rng();
  if (r < PART_RARITY_CUTS.common) return "common";
  if (r < PART_RARITY_CUTS.rare) return "rare";
  if (r < PART_RARITY_CUTS.epic) return "epic";
  return "legendary";
}

function weaponFamilies(ownedWeapons) {
  const set = new Set(["gun", "drone"]);
  for (const w of ownedWeapons || []) {
    if (w && w !== "gun" && w !== "drone") set.add(w);
  }
  return [...set];
}

function pickIndex(rng, n) {
  if (n <= 0) return 0;
  return Math.floor(rng() * n);
}

export function rollPart(rng, ownedWeapons, options = {}) {
  let family = null;
  let list = PARTS.common;
  const forced = options.family;

  if (forced === "common") {
    family = null;
    list = PARTS.common;
  } else if (forced && PARTS[forced]) {
    family = forced;
    list = PARTS[forced];
  } else if (rng() >= PART_UNIVERSAL_CHANCE) {
    const families = weaponFamilies(ownedWeapons);
    family = families[pickIndex(rng, families.length)];
    list = PARTS[family];
  }

  const def = list[pickIndex(rng, list.length)];
  const rarity = RARITY_STEP[options.rarity] ? options.rarity : rollRarity(rng);
  const part = {
    base: def.id,
    baseName: def.name,
    family,
    rarity,
    affixes: partAffixes(family, def.id, rarity),
  };
  if (options.id) part.id = options.id;
  return part;
}

function partById(parts, partId) {
  return parts.find((p) => p.id === partId);
}

function equippedBases(slots, parts) {
  const bases = new Set();
  for (const id of slots) {
    if (!id) continue;
    const p = partById(parts, id);
    if (p) bases.add(p.base);
  }
  return bases;
}

function slotFamily(slots, parts, index) {
  const id = slots[index];
  if (!id) return null;
  const p = partById(parts, id);
  return p?.family ?? null;
}

function firstCommonSlot(slots) {
  for (let i = 4; i < 8; i++) if (!slots[i]) return i;
  return -1;
}

export function firstSlot(parts, slots, partId) {
  const part = partById(parts, partId);
  if (!part) return { error: "Нет свободного слота" };
  if (equippedBases(slots, parts).has(part.base)) {
    return { error: "Такая уже надета" };
  }

  if (part.family === null) {
    const idx = firstCommonSlot(slots);
    if (idx < 0) return { error: "Нет свободного слота" };
    const next = [...slots];
    next[idx] = partId;
    return { slots: next };
  }

  for (let i = 0; i < 4; i++) {
    if (slots[i]) continue;
    const next = [...slots];
    next[i] = partId;
    return { slots: next };
  }
  const c = firstCommonSlot(slots);
  if (c < 0) return { error: "Нет свободного слота" };
  const next = [...slots];
  next[c] = partId;
  return { slots: next };
}

export function equipAt(parts, slots, partId, index) {
  const part = partById(parts, partId);
  if (!part) return { error: "Нет свободного слота" };

  if (equippedBases(slots, parts).has(part.base)) {
    return { error: "Такая уже надета" };
  }

  if (part.family === null) {
    if (index < 4) return { error: "Нет свободного слота" };
  } else if (index < 4) {
    const fam = slotFamily(slots, parts, index);
    if (fam != null && fam !== part.family) return { error: "Нет свободного слота" };
  }

  if (index < 0 || index > 7 || slots[index]) {
    return { error: "Нет свободного слота" };
  }

  const next = [...slots];
  next[index] = partId;
  return { slots: next };
}

export function unequip(slots, index) {
  const next = [...slots];
  if (index >= 0 && index < next.length) next[index] = null;
  return next;
}

const FAMILY_DMG_KEY = {
  gun: "gunDmg",
  drone: "droneDmg",
  laser: "laserDmg",
  scatter: "scatterDmg",
  grenade: "grenadeDmg",
  emp: "empDmg",
  orb: "orbDmg",
};

export const SET_TIERS = [2, 3, 4];

const WEAPON_SET_DMG = { 2: 0.08, 3: 0.16, 4: 0.28 };

const COMMON_SET = {
  2: { hp: 40 },
  3: { hp: 40, regen: 0.8 },
  4: { hp: 40, regen: 0.8, allDmg: 0.06 },
};

function highestTier(count) {
  let best = 0;
  for (const tier of SET_TIERS) if (count >= tier) best = tier;
  return best;
}

function familyCounts(parts, slots) {
  const counts = {};
  for (const id of slots || []) {
    if (!id) continue;
    const part = partById(parts, id);
    if (!part) continue;
    const key = part.family || "common";
    counts[key] = (counts[key] || 0) + 1;
  }
  return counts;
}

export function setBonuses(parts, slots) {
  const out = {};
  const counts = familyCounts(parts, slots);
  for (const [fam, count] of Object.entries(counts)) {
    const tier = highestTier(count);
    if (!tier) continue;
    if (fam === "common") {
      for (const [key, value] of Object.entries(COMMON_SET[tier])) out[key] = (out[key] || 0) + value;
    } else if (FAMILY_DMG_KEY[fam]) {
      out[FAMILY_DMG_KEY[fam]] = (out[FAMILY_DMG_KEY[fam]] || 0) + WEAPON_SET_DMG[tier];
    }
  }
  return out;
}

export function setDesc(family, tier) {
  if (!tier) return "";
  if (!family || family === "common") {
    const parts = [];
    const bonus = COMMON_SET[tier];
    if (bonus.hp) parts.push(`+${bonus.hp} HP ядра`);
    if (bonus.regen) parts.push(`+${formatAffixNumber(bonus.regen)} реген`);
    if (bonus.allDmg) parts.push(`+${Math.round(bonus.allDmg * 100)}% урона всего оружия`);
    return parts.join(", ");
  }
  return `+${Math.round(WEAPON_SET_DMG[tier] * 100)}% урона (${partFamilyLabel(family).toLowerCase()})`;
}

export function activeSets(parts, slots) {
  const counts = familyCounts(parts, slots);
  const list = [];
  for (const [fam, count] of Object.entries(counts)) {
    const tier = highestTier(count);
    if (!tier) continue;
    list.push({
      family: fam === "common" ? null : fam,
      count,
      tier,
      desc: setDesc(fam, tier),
    });
  }
  list.sort((a, b) => b.count - a.count);
  return list;
}

const HASTE_KEYS = new Set(["haste", "guardHaste", "lungeHaste"]);

function mergeBranch(into, branches) {
  into.branches = into.branches || {};
  for (const [family, byBranch] of Object.entries(branches || {})) {
    into.branches[family] = into.branches[family] || {};
    for (const [branch, mods] of Object.entries(byBranch)) {
      const dst = into.branches[family][branch] || {};
