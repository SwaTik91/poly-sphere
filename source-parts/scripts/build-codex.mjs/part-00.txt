function esc(value) {
  return String(value)
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;");
}

function readPython(name) {
  const text = fs.readFileSync(path.join(root, "server/store.py"), "utf8");
  const match = text.match(new RegExp(`^${name}\\s*=\\s*([0-9.]+)`, "m"));
  if (!match) throw new Error(`нет константы ${name}`);
  return Number(match[1]);
}

function cards(list) {
  if (!list?.length) return "<p class='muted'>Нет карт.</p>";
  return `<ul class="cards">${list
    .map(
      (card) =>
        `<li><b>${esc(card.title)}</b> <span class="tag">${esc(RARITY[card.rarity] || card.rarity || "")}</span><div>${esc(card.desc || "")}</div></li>`,
    )
    .join("")}</ul>`;
}

function stats(obj) {
  const rows = Object.entries(obj).filter(([, value]) => typeof value === "number" || typeof value === "boolean");
  if (!rows.length) return "";
  return `<table><tbody>${rows
    .map(([key, value]) => `<tr><th>${esc(STAT[key] || key)}</th><td>${esc(value)}</td></tr>`)
    .join("")}</tbody></table>`;
}

function weaponSection(weapon) {
  const info = WEAPON_INFO[weapon.id];
  const base = weapon.id === "gun" ? GUN_START : weapon.id === "drone" ? DRONE_START : defaultWep(weapon.id);
  const choice = weapon.step(5, null).cards;
  const shared = weapon.pool(null);
  const branches = choice
    .map((card) => {
      const extra = weapon.pool
        ? weapon.pool(card.id).filter((item) => !shared.some((baseCard) => baseCard.id === item.id))
        : [];
      return `<section class="branch"><h3>${esc(card.title)}</h3><p>${esc(card.desc)}</p>
        <h4>10-е улучшение</h4>${cards(weapon.step(10, card.id).cards)}
        <h4>15-е улучшение</h4>${cards(weapon.step(15, card.id).cards)}
        ${extra.length ? `<h4>Карты этой ветки в забеге</h4>${cards(extra)}` : ""}
      </section>`;
    })
    .join("");
  const finish = {
    gun: "Пробивание",
    drone: "Пробивание",
    laser: "Ширина",
    scatter: "Картечь",
    grenade: "Воронка",
    emp: "Поле",
    orb: "Рой",
  }[weapon.id];
  return `<section id="w-${weapon.id}"><h2>${esc(info.name)}</h2>${stats(base)}
    <h3>Ветки на 5-м улучшении</h3>${branches}
    <h3>Общие карты забега</h3>${cards(shared)}
    <p>После 15-го улучшения остаются «Калибр», «Темп» и «${esc(finish)}». Они копятся и дальше. Новой ветки нет.</p>
  </section>`;
}

function tierPhrase(tier) {
  if (tier.lines.length === 1) return tier.lines[0].text;
  return tier.lines.map((item) => `${item.name} ${item.text}`).join(", ");
}

function partsSection(chance, every) {
  const levelChance = readPython("LEVEL_PART_CHANCE");
  const levelPity = readPython("LEVEL_PART_PITY");
  const families = ["gun", "drone", "laser", "scatter", "grenade", "emp", "orb", "common"];
  const blocks = families
    .map((family) => {
      const items = (PARTS[family] || [])
        .map((def) => {
          const ladder = ["common", "rare", "epic", "legendary"].map((rarity) => tierPhrase(def.tiers[rarity])).join(" / ");
          const where = def.when ? `, ветка ${def.when}` : "";
          const note = def.note ? ` (${def.note})` : "";
          return `<li><b>${esc(def.name)}</b>${esc(where)}${esc(note)} — ${esc(ladder)}</li>`;
        })
        .join("");
      const title = family === "common" ? "Общие" : partFamilyLabel(family);
      return `<h3>${esc(title)}</h3><ul>${items}</ul>`;
    })
    .join("");
  const rare = Math.round((PART_RARITY_CUTS.rare - PART_RARITY_CUTS.common) * 100);
  const epic = Math.round((PART_RARITY_CUTS.epic - PART_RARITY_CUTS.rare) * 100);
  const legend = Math.round((1 - PART_RARITY_CUTS.epic) * 100);
  const setFamilies = ["gun", "drone", "laser", "scatter", "grenade", "emp", "orb", "common"];
  const setRows = setFamilies
    .map((family) => {
      const label = family === "common" ? "Общие" : partFamilyLabel(family);
      const fam = family === "common" ? null : family;
      const tiers = SET_TIERS.map((tier) => `${tier} дет.: ${esc(setDesc(fam, tier))}`).join("; ");
      return `<tr><td>${esc(label)}</td><td>${tiers}</td></tr>`;
    })
    .join("");
  return `<section id="parts"><h2>Запчасти</h2>
    <p>За пройденный уровень запчасть выпадает с шансом ${Math.round(levelChance * 100)}%; после ${levelPity} пустых подряд следующая деталь гарантирована. В бесконечном режиме бросок на каждой ${every}-й волне после старта режима, шанс ${Math.round(chance * 100)}%.</p>
    <p>Общая деталь выпадает в ${Math.round(PART_UNIVERSAL_CHANCE * 100)}% случаев, иначе семейство купленного оружия. Редкость: обычная ${Math.round(PART_RARITY_CUTS.common * 100)}%, редкая ${rare}%, эпическая ${epic}%, легендарная ${legend}%. У детали одна идея: редкость усиливает её числа. Деталь ветки молчит, пока эта ветка не выбрана, кроме строк, где сказано, что бонус работает сразу.</p>
    ${blocks}
    <h3>Сеты</h3>
    <p>Сет включается, когда надето несколько деталей одного семейства. Бонусы сета не ограничены потолками свойств.</p>
    <table><thead><tr><th>Семейство</th><th>Бонусы по числу деталей</th></tr></thead><tbody>${setRows}</tbody></table>
    <h3>Качество, полировка и пыль</h3>
    <p>Ненадетую деталь можно разобрать на пыль: обычная +${SALVAGE_DUST.common}, редкая +${SALVAGE_DUST.rare}, эпическая +${SALVAGE_DUST.epic}, легендарная +${SALVAGE_DUST.legendary}.</p>
    <p>Качество повышается только вручную. Игрок выбирает деталь и две такие же той же редкости со склада. Выбранные копии исчезают, а сама деталь становится на ранг выше: обычная, редкая, эпическая, легендарная.</p>
    <p>Легендарную деталь можно полировать до ${POLISH_MAX} раз. Полировка не повышает ранг и усиливает её собственные числа примерно на 8% за ступень. Пыль за ступени: ${POLISH_COST.join(", ")}.</p>
    <p>В магазине за монеты покупается пыль: пачка +${DUST_PACK.dust} за ${DUST_PACK.coins} монет.</p>
  </section>`;
}
