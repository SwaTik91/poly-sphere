not run_id:
                raise ValueError("Нет номера забега")
        season = season_for_date(moscow_date(now_dt))
        self.db.execute("BEGIN IMMEDIATE")
        try:
            existing = self.db.execute(
                "SELECT 1 FROM endless_claims WHERE account_id = ? AND level = ? AND wave = ?",
                (account_id, level, wave),
            ).fetchone()
            granted = 0 if existing else payout
            coins = 0
            if not existing:
                if not trusted:
                    self._require_horde_step(account_id, run_id, level, wave, now_dt)
                if horde_rule_for(moscow_day(now_dt)) == "coins":
                    coins = HORDE_RULE_COINS
                self.db.execute(
                    "INSERT INTO endless_claims (account_id, level, wave) VALUES (?, ?, ?)",
                    (account_id, level, wave),
                )
                self.db.execute(
                    "UPDATE accounts SET crystals = crystals + ?, coins = coins + ? WHERE id = ?",
                    (granted, coins, account_id),
                )
                if run_id and not trusted:
                    self._advance_horde_run(account_id, run_id, level, wave, now_dt)
            step = wave - 10 if wave >= HORDE_FIRST_WAVE else wave
            self._record_endless_progress(account_id, step, season, now_dt)
            profile = self._profile(account_id)
            self.db.commit()
        except Exception:
            self.db.rollback()
            raise
        return {"granted": granted, "crystals": profile["crystals"], "coins": coins, "profile": profile}

    def _require_horde_step(self, account_id, run_id, level, wave, now_dt):
        started = self.db.execute(
            "SELECT start_level FROM run_events WHERE account_id = ? AND run_id = ?",
            (account_id, run_id),
        ).fetchone()
        if started is None:
            raise ValueError("Забег не начат")
        cleared = self.db.execute(
            "SELECT 1 FROM run_level_clears WHERE account_id = ? AND run_id = ? AND level = ?",
            (account_id, run_id, level),
        ).fetchone()
        if cleared is None:
            raise ValueError("Уровень ещё не засчитан")
        horde = self.db.execute(
            "SELECT level, next_wave, last_claim_at FROM horde_runs WHERE account_id = ? AND run_id = ?",
            (account_id, run_id),
        ).fetchone()
        if horde is None:
            if wave != HORDE_FIRST_WAVE:
                raise ValueError("Волна ещё не открыта")
            return
        if int(horde["level"]) != int(level):
            raise ValueError("Некорректный уровень")
        if int(horde["next_wave"]) != int(wave):
            raise ValueError("Волна ещё не открыта")
        if now_dt < _parse_time(horde["last_claim_at"]) + HORDE_CLAIM_GAP:
            raise ValueError("Рано для следующей волны")

    def _advance_horde_run(self, account_id, run_id, level, wave, now_dt):
        self.db.execute(
            """
            INSERT INTO horde_runs (account_id, run_id, level, next_wave, last_claim_at)
            VALUES (?, ?, ?, ?, ?)
            ON CONFLICT(account_id, run_id) DO UPDATE SET
                next_wave = excluded.next_wave,
                last_claim_at = excluded.last_claim_at
            """,
            (account_id, run_id, int(level), int(wave) + 1, now_dt.isoformat()),
        )

    def note_level_clear(self, token, run_id, level, now=None):
        account_id = self._account_row(token)["id"]
        run_id = str(run_id or "").strip()
        if not run_id:
            raise ValueError("Нет номера забега")
        try:
            level = int(level)
        except (TypeError, ValueError) as exc:
            raise ValueError("Некорректный уровень") from exc
        if level not in range(1, MAX_LEVEL + 1):
            raise ValueError("Некорректный уровень")
        now_dt = _parse_time(now)
        self.db.execute("BEGIN IMMEDIATE")
        try:
            started = self.db.execute(
                "SELECT start_level FROM run_events WHERE account_id = ? AND run_id = ?",
                (account_id, run_id),
            ).fetchone()
            if started is None:
                raise ValueError("Забег не начат")
            start_level = int(started["start_level"] or 1)
            if level < start_level:
                raise ValueError("Некорректный уровень")
            already = self.db.execute(
                "SELECT 1 FROM run_level_clears WHERE account_id = ? AND run_id = ? AND level = ?",
                (account_id, run_id, level),
            ).fetchone()
            if already:
                profile = self._profile(account_id)
                self.db.commit()
                return {"ok": True, "profile": profile}
            if level > start_level:
                previous = self.db.execute(
                    "SELECT 1 FROM run_level_clears WHERE account_id = ? AND run_id = ? AND level = ?",
                    (account_id, run_id, level - 1),
                ).fetchone()
                if previous is None:
                    raise ValueError("Уровень ещё не засчитан")
            last = self.db.execute(
                """
                SELECT cleared_at FROM run_level_clears
                WHERE account_id = ? AND run_id = ?
                ORDER BY cleared_at DESC LIMIT 1
                """,
                (account_id, run_id),
            ).fetchone()
            if last and now_dt < _parse_time(last["cleared_at"]) + LEVEL_NOTE_GAP:
                raise ValueError("Рано для следующего уровня")
            self.db.execute(
                "INSERT INTO run_level_clears (account_id, run_id, level, cleared_at) VALUES (?, ?, ?, ?)",
                (account_id, run_id, level, now_dt.isoformat()),
            )
            profile = self._profile(account_id)
            self.db.commit()
        except Exception:
            self.db.rollback()
            raise
        return {"ok": True, "profile": profile}

    def _record_endless_progress(self, account_id, wave, season, now):
        wave = int(wave)
        if isinstance(now, str):
            now = datetime.fromisoformat(now)
        if now.tzinfo is None:
            now = now.replace(tzinfo=timezone.utc)
        self.db.execute(
            "UPDATE accounts SET best_endless = MAX(best_endless, ?) WHERE id = ?",
            (wave, account_id),
        )
        day = moscow_day(now)
        self.db.execute(
            "INSERT OR IGNORE INTO daily (account_id, day) VALUES (?, ?)",
            (account_id, day),
        )
        self.db.execute(
            "UPDATE daily SET endless_wave = MAX(endless_wave, ?) WHERE account_id = ? AND day = ?",
            (wave, account_id, day),
        )
        self.db.execute(
            """
            INSERT INTO endless_leaderboard (account_id, season, best_wave, updated_at)
            VALUES (?, ?, ?, ?)
            ON CONFLICT(account_id, season) DO UPDATE SET
                best_wave = MAX(best_wave, excluded.best_wave),
                updated_at = CASE
                    WHEN excluded.best_wave > endless_leaderboard.best_wave THEN excluded.updated_at
                    ELSE endless_leaderboard.updated_at
                END
            """,
            (account_id, season, wave, now.isoformat()),
        )

    def leaderboard(self, token, now=None):
        account_id = self._account_row(token)["id"]
        if now is None:
            now = datetime.now(timezone.utc)
        d = moscow_date(now)
        season = season_for_date(d)
        rows = self.db.execute(
            """
            SELECT a.name AS name, l.best_wave AS best_wave, l.updated_at AS updated_at, l.account_id AS account_id
            FROM endless_leaderboard l
            JOIN accounts a ON a.id = l.account_id
            WHERE l.season = ? AND l.best_wave > 0
            ORDER BY l.best_wave DESC, l.updated_at ASC
            """,
            (season,),
        ).fetchall()
        top = [
            {"rank": index + 1, "name": row["name"], "wave": int(row["best_wave"])}
            for index, row in enumerate(rows[:LEADERBOARD_SIZE])
        ]
        me = None
        for index, row in enumerate(rows):
            if row["account_id"] == account_id:
                me = {"rank": index + 1, "wave": int(row["best_wave"])}
                break
        return {
            "season": {
                "id": season,
                "label": season_label(season),
                "daysLeft": season_days_left(d),
            },
            "top": top,
            "me": me,
        }

    def buy_reroll(self, token, run_id, offer_level):
        run_id = str(run_id or "").strip()
        try:
            offer_level = int(offer_level)
        except (TypeError, ValueError):
            raise ValueError("Некорректный реролл")
        if not run_id or offer_level < 1:
            raise ValueError("Некорректный реролл")
        account_id = self._account_row(token)["id"]
        self.db.execute("BEGIN IMMEDIATE")
        try:
            row = self.db.execute(
                "SELECT used FROM reroll_counts WHERE account_id = ? AND run_id = ? AND offer_level = ?",
                (account_id, run_id, offer_level),
            ).fetchone()
            used = int(row["used"]) if row else 0
            price = reroll_price(used)
            if price is None:
                raise ValueError("Реролл больше недоступен")
            account = self.db.execute("SELECT coins, crystals FROM accounts WHERE id = ?", (account_id,)).fetchone()
            if price["kind"] == "coins" and account["coins"] < price["amount"]:
                raise ValueError("Не хватает монет")
            if price["kind"] == "crystals" and account["crystals"] < price["amount"]:
                raise ValueError("Не хватает кристаллов")
            if price["kind"] == "coins":
                self.db.execute("UPDATE accounts SET coins = coins - ? WHERE id = ?", (price["amount"], account_id))
            elif price["kind"] == "crystals":
                self.db.execute(
                    "UPDATE accounts SET crystals = crystals - ? WHERE id = ?",
                    (price["amount"], account_id),
                )
            used += 1
            self.db.execute(
                """
                INSERT INTO reroll_counts (account_id, run_id, offer_level, used)
                VALUES (?, ?, ?, ?)
                ON CONFLICT(account_id, run_id, offer_level) DO UPDATE SET used = excluded.used
                """,
                (account_id, run_id, offer_level, used),
            )
            profile = self._profile(account_id)
            self.db.commit()
        except Exception:
            self.db.rollback()
            raise
        return {"used": used, "price": price, "next": reroll_price(used), "profile": profile}

    def buy_hangar(self, token, key):
        if key not in HANGAR_TREE:
            raise ValueError("Неизвестное улучшение")
        return self._buy_hangar_node(token, key)

    def _hangar_node_level(self, account_id, key):
        node_row = self.db.execute(
            "SELECT level FROM account_hangar_nodes WHERE account_id = ? AND node = ?",
            (account_id, key),
        ).fetchone()
        return int(node_row["level"]) if node_row else 0

    def _buy_hangar_node(self, token, key):
        account_id = self._account_row(token)["id"]
        spec = HANGAR_TREE[key]
        self.db.execute("BEGIN IMMEDIATE")
        try:
            row = self.db.execute("SELECT * FROM accounts WHERE id = ?", (account_id,)).fetchone()
            requires = spec["requires"]
            if requires:
                req_node, req_level = requires
                if self._hangar_node_level(account_id, req_node) < req_level:
                    raise ValueError("Ветка ещё закрыта")
            level = self._hangar_node_level(account_id, key)
            if level >= spec["max"]:
                raise ValueError("Уже максимум")
            price = hangar_node_price(key, level)
            if row["coins"] < price:
                raise ValueError("Не хватает монет")
            self.db.execute(
                "UPDATE accounts SET coins = coins - ?, hangar_buys = hangar_buys + 1 WHERE id = ?",
                (price, account_id),
            )
            self.db.execute(
                """
                INSERT INTO account_hangar_nodes (account_id, node, level)
                VALUES (?, ?, 1)
                ON CONFLICT(account_id, node) DO UPDATE SET level = level + 1
                """,
                (account_id, key),
            )
            self.db.commit()
        except Exception:
            self.db.rollback()
            raise
        return {"profile": self._profile(account_id)}

    def _weapon_loadout(self, account_id, row=None):
        if row is None:
            row = self.db.execute(
                "SELECT weapon_loadout_json FROM accounts WHERE id = ?", (account_id,)
            ).fetchone()
        raw = row["weapon_loadout_json"] if row is not None else None
        try:
            slots = json.loads(raw) if raw else list(DEFAULT_WEAPON_LOADOUT)
        except (TypeError, ValueError):
            slots = list(DEFAULT_WEAPON_LOADOUT)
        return self._sanitize_weapon_loadout(account_id, slots)

    def _sanitize_weapon_loadout(self, account_id, loadout):
        owned = set(self._weapons(account_id)) | set(FREE_WEAPONS)
        slots = [None] * WEAPON_SLOTS
        slots[0] = "gun"
        seen = {"gun"}
        idx = 1
        for item in loadout or []:
            if idx >= WEAPON_SLOTS:
                break
            if item in (None, "", "gun"):
                continue
            if item not in ALL_WEAPONS:
                raise ValueError("Нет такого орудия")
            if item not in owned:
                raise ValueError("Орудие не куплено")
            if item in seen:
                continue
            slots[idx] = item
            seen.add(item)
            idx += 1
        return slots

        try:
            self.db.execute(

            )
            profile = self._profile(account_id)
            self.db.commit()
        except Exception:
            self.db.rollback()
            raise
        return {"profile": profile}
