    def roll_part(self, token, run_id, kind, level, wave, rng, roller=None):
        row = self._account_row(token)
        account_id = row["id"]
        tester = dev_tools_enabled() and is_tester_name(row["name"])
        run_id = str(run_id or "").strip()
        kind = str(kind or "").strip()
        if not run_id:
            raise ValueError("Нет номера забега")
        if kind not in ("level", "endless"):
            raise ValueError("Некорректный бросок детали")
        self.db.execute("BEGIN IMMEDIATE")
        try:
            if kind == "endless":
                try:
                    wave_num = int(wave)
                except (TypeError, ValueError):
                    wave_num = 0
                if wave_num <= 0 or wave_num % ENDLESS_PART_EVERY != 0:
                    profile = self._profile(account_id)
                    self.db.commit()
                    return {"part": None, "profile": profile}
                roll_key = f"{run_id}:endless:{wave_num}"
            else:
                try:
                    level_num = int(level)
                except (TypeError, ValueError):
                    raise ValueError("Некорректный бросок детали")
                roll_key = f"{run_id}:level:{level_num}"

            existing = self.db.execute(
                "SELECT part_id FROM part_rolls WHERE account_id = ? AND roll_key = ?",
                (account_id, roll_key),
            ).fetchone()
            if existing:
                part_id = existing["part_id"]
                part = self._part_by_id(account_id, part_id) if part_id else None
                profile = self._profile(account_id)
                self.db.commit()
                return {"part": part, "profile": profile}

            if kind == "level":
                counted = self.db.execute(
                    "SELECT 1 FROM run_level_clears WHERE account_id = ? AND run_id = ? AND level = ?",
                    (account_id, run_id, level_num),
                ).fetchone()
                if counted is None:
                    raise ValueError("Уровень ещё не засчитан")
            else:
                try:
                    level_num = int(level)
                except (TypeError, ValueError) as exc:
                    raise ValueError("Волна ещё не засчитана") from exc
                counted = self.db.execute(
                    "SELECT 1 FROM endless_claims WHERE account_id = ? AND level = ? AND wave = ?",
                    (account_id, level_num, 10 + wave_num),
                ).fetchone()
                if counted is None:
                    raise ValueError("Волна ещё не засчитана")

            drop = False
            if kind == "level":
                dry = int(row["parts_dry"] or 0)
                if tester or dry >= LEVEL_PART_PITY or rng() < LEVEL_PART_CHANCE:
                    drop = True
                else:
                    self._record_roll_miss(account_id, roll_key, kind="level")
                    profile = self._profile(account_id)
                    self.db.commit()
                    return {"part": None, "profile": profile}
            elif rng() < ENDLESS_PART_CHANCE:
                drop = True
            else:
                self._record_roll_miss(account_id, roll_key, kind="endless")
                profile = self._profile(account_id)
                self.db.commit()
                return {"part": None, "profile": profile}

            owned = self._owned_weapon_families(account_id)
            part_id = secrets.token_hex(16)
            if roller is None:
                part = parts_bridge.roll_part(rng, owned, part_id)
            else:
                part = roller(rng, owned)
                if not part.get("id"):
                    part = {**part, "id": part_id}
            self._insert_part(account_id, part)
            self.db.execute(
                "INSERT INTO part_rolls (account_id, roll_key, part_id) VALUES (?, ?, ?)",
                (account_id, roll_key, part["id"]),
            )
            if kind == "level":
                self.db.execute("UPDATE accounts SET parts_dry = 0 WHERE id = ?", (account_id,))
            profile = self._profile(account_id)
            self.db.commit()
        except Exception:
            self.db.rollback()
            raise
        return {"part": part, "profile": profile}

    def upgrade_part(self, token, part_id, fodder_ids):
        part_id = str(part_id or "").strip()
        fodder = []
        for raw in fodder_ids or []:
            item = str(raw or "").strip()
            if item and item not in fodder:
                fodder.append(item)
        if not part_id:
            raise ValueError("Нет детали")
        if len(fodder) != 2 or part_id in fodder:
            raise ValueError("Выберите две такие же детали")
        account_id = self._account_row(token)["id"]
        self.db.execute("BEGIN IMMEDIATE")
        try:
            part = self._part_by_id(account_id, part_id)
            if part is None:
                raise ValueError("Нет детали")
            idx = PART_RARITY_ORDER.index(part["rarity"]) if part["rarity"] in PART_RARITY_ORDER else -1
            if idx < 0 or idx >= len(PART_RARITY_ORDER) - 1:
                raise ValueError("Уже максимум")
            next_rarity = PART_RARITY_ORDER[idx + 1]
            row = self.db.execute("SELECT loadout_json FROM accounts WHERE id = ?", (account_id,)).fetchone()
            slots = json.loads(row["loadout_json"] or DEFAULT_LOADOUT_JSON)
            for fodder_id in fodder:
                if fodder_id in slots:
                    raise ValueError("Деталь надета")
                other = self._part_by_id(account_id, fodder_id)
                if other is None:
                    raise ValueError("Нет детали")
                same = (
                    other["base"] == part["base"]
                    and other["rarity"] == part["rarity"]
                    and (other["family"] or None) == (part["family"] or None)
                )
                if not same:
                    raise ValueError("Нужны такие же детали")
            rebuilt = parts_bridge.part_affixes(part.get("family"), part["base"], next_rarity)
            if rebuilt:
                affixes = rebuilt
            else:
                step = PART_RARITY_STEP[next_rarity]
                affixes = []
                for affix in part["affixes"]:
                    updated = dict(affix)
                    updated["step"] = step
                    affixes.append(updated)
            for fodder_id in fodder:
                self.db.execute(
                    "DELETE FROM account_parts WHERE account_id = ? AND id = ?",
                    (account_id, fodder_id),
                )
            self.db.execute(
                "UPDATE account_parts SET rarity = ?, affixes_json = ?, polish = 0 WHERE account_id = ? AND id = ?",
                (next_rarity, json.dumps(affixes, ensure_ascii=False), account_id, part_id),
            )
            profile = self._profile(account_id)
            self.db.commit()
        except Exception:
            self.db.rollback()
            raise
        return {"profile": profile, "part": self._part_by_id(account_id, part_id)}

    def polish_part(self, token, part_id):
        part_id = str(part_id or "").strip()
        if not part_id:
            raise ValueError("Нет детали")
        account_id = self._account_row(token)["id"]
        self.db.execute("BEGIN IMMEDIATE")

            if part is None:
                raise ValueError("Нет детали")
            if part["rarity"] != "legendary":
                raise ValueError("Полировать можно только легендарную")
            level = int(part.get("polish") or 0)
            if level >= POLISH_MAX:
                raise ValueError("Полировка на максимуме")
            cost = POLISH_COST[level]
            account = self.db.execute("SELECT dust FROM accounts WHERE id = ?", (account_id,)).fetchone()
            if int(account["dust"] or 0) < cost:
                raise ValueError("Мало пыли")
            self.db.execute(
                "UPDATE account_parts SET polish = ? WHERE account_id = ? AND id = ?",
                (level + 1, account_id, part_id),
            )
            self.db.execute(
                "UPDATE accounts SET dust = dust - ? WHERE id = ?",
                (cost, account_id),
            )
            profile = self._profile(account_id)
            self.db.commit()
        except Exception:
            self.db.rollback()
            raise
        return {"profile": profile, "part": self._part_by_id(account_id, part_id)}
